-- //==========// IDENTIFICA SE BIY_HATCH CHEGOU AO SENSOR //==========//

-- Declara as variaveis locais
var r, p: integer                              -- Inteiros para os Loops
var robo, ponto_solda: object                  -- Define o robo e ponto atual da logica
var tabela_tempos, tabela_coords: object       -- Define as tabelas a serem usadas

-- Se MU for HATCH
if @.name = "BIY_HATCH" then
	-- Tabela usada = tabela HATCH
	tabela_coords := TC_HATCH;

-- Se MU for SEDAN
elseif @.name = "BIY_SEDAN" then
	-- Tabela usada = tabela SEDAN
	tabela_coords := TC_SEDAN;
end

str_modeloAtual := @.name

-- Atualiza as tabelas de tempos dos quatro robos
for r := 1 to 4 loop
	-- Referencia do robo
	robo := str_to_obj("ABB_IRB_6700_" + to_str(r));

	-- Tabela a ser usada
	if @.name = "BIY_HATCH" then
		tabela_tempos := str_to_obj("TT_R" + to_str(r) + "_HATCH");
	elseif @.name = "BIY_SEDAN" then
		tabela_tempos := str_to_obj("TT_R" + to_str(r) + "_SEDAN");
	end

	-- Define a tabela de tempos do robo
	robo.setTimesTable(tabela_tempos);
next

-- Laco para os robos
for r := 1 to 4 loop
	-- Laco para os pontos
	for p := 1 to 10 loop
		-- Referencia do ponto de solda
		ponto_solda := str_to_obj("R" + to_str(r) + "_P" + to_str(p));

		-- Define as coordenadas do ponto
		ponto_solda._3D.Position.X := tabela_coords["X_R" + to_str(r), "P" + to_str(p)];
		ponto_solda._3D.Position.Y := tabela_coords["Y_R" + to_str(r), "P" + to_str(p)];
		ponto_solda._3D.Position.Z := tabela_coords["Z_R" + to_str(r), "P" + to_str(p)];
	next
next

-- Apos definir o BIY e os pontos, para o SKID
SKID_EFQ_3.startPause;

-- Laco externo para P (sources -> pontos impares)
-- Um determinado inteiro impar equivale ao dobro de sua posicao na sequencia subtraido de um
-- Comeca criando o primeiro ponto de solda de cada robo para tornar o processo mais sincrono
for p := 1 to 5 loop
    -- Laco interno para os robos (R1 a R4)
    for r := 1 to 4 loop
        -- create -> cria o MU (SOLDA) no objeto [R(r)_P(2*p - 1)]
        ~.~.UserObjects.SOLDA.create(str_to_obj("R" + to_str(r) + "_P" + to_str(2*p - 1)));
    next
next

-- Espera ate que o robo R1 esteja no ultimo ponto
waituntil (ABB_IRB_6700_1.resWorking = false and int_pontoR1 = 9) prio 1;

-- Espera ate que o robo R1 esteja se movendo para a posicao de repouso
waituntil (ABB_IRB_6700_1.resWorking = true and int_pontoR1 = 0) prio 1;

-- Espera ate que todos os robos estejam na posicao de repouso
waituntil (ABB_IRB_6700_1.resWorking = false and int_pontoR1 = 0) and
          (ABB_IRB_6700_2.resWorking = false and int_pontoR2 = 0) and
          (ABB_IRB_6700_3.resWorking = false and int_pontoR3 = 0) and
          (ABB_IRB_6700_4.resWorking = false and int_pontoR4 = 0) prio 1;

-- Ativa o SKID
SKID_EFQ_3.pause = false;
